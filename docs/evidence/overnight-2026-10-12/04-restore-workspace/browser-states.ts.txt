import fs from "node:fs";
import path from "node:path";
import { chromium, expect } from "@playwright/test";
import AxeBuilder from "@axe-core/playwright";
import { DatabaseSync } from "node:sqlite";
async function main() {
  const run="overnight-final-ui";
  const manifest=JSON.parse(fs.readFileSync(`scratch/qa-${run}/manifest.json`,"utf8"));
  if(manifest.outbound!=="stubbed" || !manifest.protectedUnchanged || manifest.port!==3107) throw new Error("Isolated QA manifest required");
  const output="docs/evidence/overnight-2026-10-12/11-browser-states";
  if(fs.existsSync(output)) throw new Error("Do not overwrite evidence");
  fs.mkdirSync(output,{recursive:true});
  const browser=await chromium.launch();
  const results:Record<string,unknown>[]=[];
  const context=await browser.newContext({viewport:{width:390,height:900},reducedMotion:"reduce"}); const page=await context.newPage();
  try {
    const response=await page.goto("http://localhost:3107/");
    expect(response?.headers()["x-robots-tag"]).toBe("noindex, nofollow");
    await page.getByRole("button",{name:"Mở menu",exact:true}).click();
    const dialog=page.getByRole("dialog",{name:"Menu điều hướng"});
    await expect(dialog).toBeVisible();
    for(let index=0;index<18;index++) {
      await page.keyboard.press("Tab");
      expect(await dialog.evaluate(element=>element.contains(document.activeElement))).toBe(true);
    }
    const menuAxe=await new AxeBuilder({page}).withTags(["wcag2a","wcag2aa","wcag21aa","wcag22aa"]).analyze();
    fs.writeFileSync(path.join(output,"menu-axe.json"),JSON.stringify(menuAxe,null,2));
    await page.screenshot({path:path.join(output,"menu-390.png")});
    await page.keyboard.press("Escape");
    await expect(dialog).not.toBeVisible();
    await expect(page.getByRole("button",{name:"Mở menu",exact:true})).toBeFocused();
    results.push({case:"menu Tab containment 18 steps + Escape returns focus",result:"pass",axeViolations:menuAxe.violations.map(v=>v.id)});
    await page.getByRole("button",{name:"Đăng ký tư vấn miễn phí",exact:true}).click();
    await expect(page.locator("#fullName-error")).toBeVisible();
    await expect(page.locator("#consent-error")).toBeVisible();
    await page.screenshot({path:path.join(output,"form-errors-390.png"),fullPage:true});
    results.push({case:"empty form reports field and consent errors",result:"pass"});
    await page.locator("#fullName").fill("QA FIXTURE ONLY");
    await page.locator("#phone").fill("0900000000");
    await page.locator("#email").fill("qa@example.invalid");
    await page.locator("#program").selectOption({index:1});
    await page.locator("#educationLevel").selectOption({index:1});
    await page.locator("#consent").check();
    await page.route("**/api/public/lead",route=>route.fulfill({status:503,contentType:"application/json",body:JSON.stringify({error:"Fixture server failure"})}));
    await page.getByRole("button",{name:"Đăng ký tư vấn miễn phí",exact:true}).click();
    await expect(page.getByRole("alert").filter({hasText:"Fixture server failure"})).toContainText("Fixture server failure");
    await expect(page.locator("#fullName")).toHaveValue("QA FIXTURE ONLY");
    await expect(page.locator("#phone")).toHaveValue("0900000000");
    await page.screenshot({path:path.join(output,"form-server-failure-390.png"),fullPage:true});
    results.push({case:"503 does not show success and preserves input",result:"pass",network:"intercepted fixture failure"});
    await page.unroute("**/api/public/lead");
    const saved=page.waitForResponse(response=>response.url().endsWith("/api/public/lead") && response.request().method()==="POST");
    await page.getByRole("button",{name:"Đăng ký tư vấn miễn phí",exact:true}).click();
    const savedResponse=await saved;
    expect(savedResponse.status()).toBe(200);
    expect((await savedResponse.json()).success).toBe(true);
    await expect(page.getByText("Gửi yêu cầu tư vấn thành công!",{exact:true})).toBeVisible();
    await page.screenshot({path:path.join(output,"form-success-fixture-390.png"),fullPage:true});
    const db=new DatabaseSync(path.resolve(`scratch/qa-${run}/fixture.db`),{readOnly:true});
    try {expect(Number(db.prepare("SELECT count(*) AS count FROM leads WHERE fullname='QA FIXTURE ONLY'").get()?.count)).toBe(1);} finally {db.close();}
    results.push({case:"success follows actual fixture API save; one synthetic lead persisted",result:"pass",outbound:"stubbed; no external notification"});
    for(const width of [320,390]) {
      await page.setViewportSize({width,height:900});
      await page.goto("http://localhost:3107/cong-nghe-thong-tin/");
      const buttons=page.locator('aside[aria-label="Hành động tuyển sinh"] a');
      for(let index=0;index<await buttons.count();index++) {
        const metrics=await buttons.nth(index).evaluate(element=>{
          const box=element.getBoundingClientRect();
          const range=document.createRange();range.selectNodeContents(element);
          const rects=Array.from(range.getClientRects()).filter(rect=>rect.width>0&&rect.height>0);
          return {left:box.left,right:box.right,height:box.height,viewport:innerWidth,textFits:rects.every(rect=>rect.left>=box.left-1&&rect.right<=box.right+1)};
        });
        expect(metrics.left).toBeGreaterThanOrEqual(0);expect(metrics.right).toBeLessThanOrEqual(width);
        expect(metrics.height).toBeGreaterThanOrEqual(44);expect(metrics.textFits).toBe(true);
        results.push({case:"sticky text and target fit",width,index,result:"pass",metrics});
      }
      await page.screenshot({path:path.join(output,`sticky-${width}.png`)});
    }
  } catch(error) {results.push({result:"fail",error:String(error)});process.exitCode=1;}
  finally {fs.writeFileSync(path.join(output,"results.json"),JSON.stringify({run,buildId:manifest.buildId,sourceFingerprint:manifest.sourceFingerprint,browser:browser.version(),capturedAt:new Date().toISOString(),dataset:"Synthetic fixture only; no research participants",results},null,2));await browser.close();}
  console.log(JSON.stringify(results));
}
main().catch(error=>{console.error(error);process.exitCode=1;});
